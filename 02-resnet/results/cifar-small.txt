  python3 resnet_cifar.py --n 1 --plain --limit 5000 --epochs 3 --seed 0 --device cpu   # plain-8

  python3 resnet_cifar.py --n 3 --plain --limit 5000 --epochs 3 --seed 0 --device cpu   # plain-20

  python3 resnet_cifar.py --n 9 --plain --limit 2000 --epochs 2 --seed 0 --device cpu   # plain-56
